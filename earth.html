import cv2 import numpy as np import base64 # Load the image img_path = 'Screenshot 2026-08-04 1.26.45 PM.jpg' img = cv2.imread(img_path) if img is not None: # Convert image to HSV color space hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV) # Define lower and upper bounds for red color (lines) # Red hues wrap around in HSV (0-10 and 160-180) lower_red1 = np.array([0, 100, 100]) upper_red1 = np.array([10, 255, 255]) lower_red2 = np.array([160, 100, 100]) upper_red2 = np.array([180, 255, 255]) # Create masks for red color mask1 = cv2.inRange(hsv, lower_red1, upper_red1) mask2 = cv2.inRange(hsv, lower_red2, upper_red2) mask = mask1 + mask2 # Dilate the mask slightly to cover anti-aliased edges of the lines kernel = np.ones((3, 3), np.uint8) mask_dilated = cv2.dilate(mask, kernel, iterations=1) # Use inpainting to fill the masked areas with surrounding pixels cleaned_img = cv2.inpaint(img, mask_dilated, 3, cv2.INPAINT_TELEA) # Encode the cleaned image to Base64 to embed directly in HTML _, buffer = cv2.imencode('.jpg', cleaned_img) jpg_as_text = base64.b64encode(buffer).decode('utf-8') # Generate HTML content html_content = f""" Cleaned Map Background

지도 배경 HTML

요청하신 대로 빨간 선을 지운 이미지를 배경으로 적용했습니다.

""" # Save the HTML file file_path = 'cleaned_map_background.html' with open(file_path, 'w', encoding='utf-8') as f: f.write(html_content) print(f"HTML file created successfully.") else: print("Failed to load image.")
